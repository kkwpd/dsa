#include <GL/glut.h>
#include <iostream>
using namespace std;

float x[4], y[4]; // 4 control points

// Draw Axis
void drawAxis() {
    glColor3f(0,1,0);
    glBegin(GL_LINES);
        glVertex2i(-300,0);
        glVertex2i(300,0);
        glVertex2i(0,-300);
        glVertex2i(0,300);
    glEnd();
}

// Bezier Function
void drawBezier() {
    float xt, yt;

    glColor3f(1,0,0); // curve color
    glBegin(GL_LINE_STRIP);

    for(float t = 0.0; t <= 1.0; t += 0.001) {

        xt = (1-t)*(1-t)*(1-t)*x[0] +
             3*t*(1-t)*(1-t)*x[1] +
             3*t*t*(1-t)*x[2] +
             t*t*t*x[3];

        yt = (1-t)*(1-t)*(1-t)*y[0] +
             3*t*(1-t)*(1-t)*y[1] +
             3*t*t*(1-t)*y[2] +
             t*t*t*y[3];

        glVertex2f(xt, yt);
    }

    glEnd();
}

// Display
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis();

    // Draw control points
    glColor3f(1,1,1);
    glPointSize(5);
    glBegin(GL_POINTS);
    for(int i=0;i<4;i++)
        glVertex2f(x[i], y[i]);
    glEnd();

    drawBezier();

    glFlush();
}

// Init
void init() {
    glClearColor(0,0,0,1);
    gluOrtho2D(-300,300,-300,300);
}

// Main
int main(int argc, char** argv) {

    cout << "Enter 4 control points:\n";
    for(int i=0;i<4;i++) {
        cin >> x[i] >> y[i];
    }

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600,600);
    glutCreateWindow("Bezier Curve");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
