#include <GL/glut.h>
#include <iostream>
using namespace std;

int xc, yc;     // Center of circle
int radius;     // Radius

// Plot a single pixel
void plotPoint(int x, int y) {
    glBegin(GL_POINTS);
    glVertex2i(x, y);
    glEnd();
}

// Draw Axis
void drawAxis() {
    glColor3f(0, 1, 0); // green

    glBegin(GL_LINES);
        glVertex2i(-500, 0);
        glVertex2i(500, 0);
        glVertex2i(0, -500);
        glVertex2i(0, 500);
    glEnd();
}

// Bresenham Circle Algorithm
void drawCircle(int xc, int yc, int r) {
    int x = 0;
    int y = r;
    int d = 3 - 2 * r;

    while (x <= y) {
        // 8-way symmetry
        plotPoint(xc + x, yc + y);
        plotPoint(xc - x, yc + y);
        plotPoint(xc + x, yc - y);
        plotPoint(xc - x, yc - y);
        plotPoint(xc + y, yc + x);
        plotPoint(xc - y, yc + x);
        plotPoint(xc + y, yc - x);
        plotPoint(xc - y, yc - x);

        if (d < 0) {
            d += 4 * x + 6;
        } else {
            d += 4 * (x - y) + 10;
            y--;
        }
        x++;
    }
}

// Display function
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis(); // axis

    glColor3f(1.0, 0.0, 0.0); // red circle
    drawCircle(xc, yc, radius);

    glFlush();
}

// Initialization
void init() {
    glClearColor(0.0, 0.0, 0.0, 1.0);
    glPointSize(2.0);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(-500, 500, -500, 500);
}

// Main function
int main(int argc, char** argv) {

    cout << "Enter center (xc yc): ";
    cin >> xc >> yc;

    cout << "Enter radius: ";
    cin >> radius;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600, 600);
    glutCreateWindow("Bresenham Circle");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
