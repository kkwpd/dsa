#include <windows.h>
#include <GL/glut.h>
#include <math.h>

float tx = 0;

void circle()
{
    glBegin(GL_POLYGON);
    for (int i = 0; i < 100; i++)
    {
        float t = 2.0f * 3.1416f * i / 100;
        glVertex2f(0.3f * cos(t), 0.3f * sin(t));
    }
    glEnd();
}

void display()
{
    glClear(GL_COLOR_BUFFER_BIT);

    glLoadIdentity();
    glTranslatef(tx, 0, 0);
    circle();

    glFlush();
}

void update(int v)
{
    tx += 0.01;
    if (tx > 1) tx = -1;

    glutPostRedisplay();
    glutTimerFunc(16, update, 0);
}

void init()
{
    glClearColor(0, 1, 1, 1);
    gluOrtho2D(-1, 1, -1, 1);
}

int main(int argc, char** argv)
{
    glutInit(&argc, argv);
    glutCreateWindow("Test");
    glutInitWindowSize(800, 600);

    init();
    glutDisplayFunc(display);
    glutTimerFunc(0, update, 0);

    glutMainLoop();
}
