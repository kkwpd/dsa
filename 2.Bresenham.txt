#include <GL/glut.h>
#include <iostream>
using namespace std;

int x1, y1, x2, y2;

// Pixel
void drawPixel(int x, int y) {
    glBegin(GL_POINTS);
    glVertex2i(x, y);
    glEnd();
}

// Axis
void drawAxis() {
    glColor3f(0,0,0);
    glBegin(GL_LINES);
        glVertex2i(-500,0);
        glVertex2i(500,0);
        glVertex2i(0,-500);
        glVertex2i(0,500);
    glEnd();
}

// Bresenham
void bresenhamLine() {
    int dx = abs(x2 - x1);
    int dy = abs(y2 - y1);

    int sx = (x2 > x1) ? 1 : -1;
    int sy = (y2 > y1) ? 1 : -1;

    int err = dx - dy;

    int x = x1;
    int y = y1;

    while(true) {
        drawPixel(x, y);

        if(x == x2 && y == y2) break;

        int e2 = 2 * err;

        if(e2 > -dy) {
            err -= dy;
            x += sx;
        }

        if(e2 < dx) {
            err += dx;
            y += sy;
        }
    }
}

// Display
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis();

    glColor3f(1,0,0);
    bresenhamLine();

    glFlush();
}

void init() {
    glClearColor(1,1,1,1);
    gluOrtho2D(-500,500,-500,500);
}

int main(int argc, char** argv) {

    cout << "Enter x1 y1: ";
    cin >> x1 >> y1;

    cout << "Enter x2 y2: ";
    cin >> x2 >> y2;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600,600);
    glutCreateWindow("Bresenham");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
