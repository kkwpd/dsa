#include <GL/glut.h>
#include <iostream>
using namespace std;

int x1, y1;

// Axis
void drawAxis() {
    glColor3f(0,1,0);
    glBegin(GL_LINES);
        glVertex2i(-500,0);
        glVertex2i(500,0);
        glVertex2i(0,-500);
        glVertex2i(0,500);
    glEnd();
}

// Polygon
void drawPolygon() {
    glColor3f(1,1,1);
    glBegin(GL_LINE_LOOP);
        glVertex2i(100,100);
        glVertex2i(200,100);
        glVertex2i(200,200);
        glVertex2i(100,200);
    glEnd();
}

// Get pixel
void getPixel(int x, int y, float color[3]) {
    glReadPixels(x,y,1,1,GL_RGB,GL_FLOAT,color);
}

// Set pixel
void setPixel(int x, int y) {
    glBegin(GL_POINTS);
    glVertex2i(x,y);
    glEnd();
}

// Boundary Fill
void boundaryFill(int x, int y, float boundaryColor[3]) {
    float color[3];
    getPixel(x,y,color);

    if(!(color[0]==boundaryColor[0] && color[1]==boundaryColor[1] && color[2]==boundaryColor[2])) {

        glColor3f(1,0,0);
        setPixel(x,y);

        boundaryFill(x+1,y,boundaryColor);
        boundaryFill(x-1,y,boundaryColor);
        boundaryFill(x,y+1,boundaryColor);
        boundaryFill(x,y-1,boundaryColor);
    }
}

// Display
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis();
    drawPolygon();

    float boundaryColor[3] = {1,1,1}; // white boundary
    boundaryFill(x1,y1,boundaryColor);

    glFlush();
}

void init() {
    glClearColor(0,0,0,1);
    gluOrtho2D(-500,500,-500,500);
}

int main(int argc, char** argv) {

    cout << "Enter seed point: ";
    cin >> x1 >> y1;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600,600);
    glutCreateWindow("Boundary Fill");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
