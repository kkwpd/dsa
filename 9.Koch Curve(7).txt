#include <GL/glut.h>
#include <math.h>
#include <iostream>
using namespace std;

int level;

// Draw Axis
void drawAxis() {
    glColor3f(0,1,0);
    glBegin(GL_LINES);
        glVertex2i(-300,0);
        glVertex2i(300,0);
        glVertex2i(0,-300);
        glVertex2i(0,300);
    glEnd();
}

// Koch Function
void koch(float x1, float y1, float x2, float y2, int n) {

    if(n == 0) {
        glVertex2f(x1, y1);
        glVertex2f(x2, y2);
        return;
    }

    float xA = x1 + (x2 - x1)/3;
    float yA = y1 + (y2 - y1)/3;

    float xB = x1 + 2*(x2 - x1)/3;
    float yB = y1 + 2*(y2 - y1)/3;

    float xC = (xA + xB)/2 + (yA - yB)/sqrt(3);
    float yC = (yA + yB)/2 + (xB - xA)/sqrt(3);

    koch(x1, y1, xA, yA, n-1);
    koch(xA, yA, xC, yC, n-1);
    koch(xC, yC, xB, yB, n-1);
    koch(xB, yB, x2, y2, n-1);
}

// Display
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis();

    glColor3f(1,0,0);

    glBegin(GL_LINES);
        koch(-200, 0, 200, 0, level);
    glEnd();

    glFlush();
}

// Init
void init() {
    glClearColor(0,0,0,1);
    gluOrtho2D(-300,300,-300,300);
}

// Main
int main(int argc, char** argv) {

    cout << "Enter level (0-4): ";
    cin >> level;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600,600);
    glutCreateWindow("Koch Curve");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
