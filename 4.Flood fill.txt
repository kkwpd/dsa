#include <GL/glut.h>
#include <iostream>
using namespace std;

int x1, y1; // seed point

// Axis
void drawAxis() {
    glColor3f(0,1,0);
    glBegin(GL_LINES);
        glVertex2i(-500,0);
        glVertex2i(500,0);
        glVertex2i(0,-500);
        glVertex2i(0,500);
    glEnd();
}

// Simple polygon (square)
void drawPolygon() {
    glColor3f(1,1,1);
    glBegin(GL_LINE_LOOP);
        glVertex2i(100,100);
        glVertex2i(200,100);
        glVertex2i(200,200);
        glVertex2i(100,200);
    glEnd();
}

// Get pixel color
void getPixel(int x, int y, float color[3]) {
    glReadPixels(x, y, 1, 1, GL_RGB, GL_FLOAT, color);
}

// Set pixel color
void setPixel(int x, int y) {
    glBegin(GL_POINTS);
    glVertex2i(x,y);
    glEnd();
}

// Flood Fill
void floodFill(int x, int y, float oldColor[3]) {
    float color[3];
    getPixel(x,y,color);

    if(color[0]==oldColor[0] && color[1]==oldColor[1] && color[2]==oldColor[2]) {
        glColor3f(1,0,0);
        setPixel(x,y);

        floodFill(x+1,y,oldColor);
        floodFill(x-1,y,oldColor);
        floodFill(x,y+1,oldColor);
        floodFill(x,y-1,oldColor);
    }
}

// Display
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    drawAxis();
    drawPolygon();

    float oldColor[3] = {0,0,0}; // background black
    floodFill(x1,y1,oldColor);

    glFlush();
}

void init() {
    glClearColor(0,0,0,1);
    gluOrtho2D(-500,500,-500,500);
}

// Main
int main(int argc, char** argv) {

    cout << "Enter seed point (inside polygon): ";
    cin >> x1 >> y1;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(600,600);
    glutCreateWindow("Flood Fill");

    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
