prompt macro msg
lea dx,msg
mov ah,09h
int 21h
endm

data segment
msg0 db 10,13," $"
msg1 db 10,13,"Hello World !$"
msg2 db 10,13,"Enter a Number : $"
msg3 db 10,13,"Sum of all Numbers : $"
msg4 db 10,13,"Avg of all Numbers : $"
msg5 db 10,13,"Multiplication of the 2 Numbers (n1 & n2): $"
no db 05h dup(0)
n1 db 00h
n2 db 00h
a db 05h
b db 04h
c db 00h
d db 00h
data ends

code segment
assume cs:code,ds:data
start:
mov ax,data
mov ds,ax
prompt msg1

prompt msg0
mov si,00h
up1:
prompt msg2
call take
mov [no+si],bl
inc si
dec [a]
jnz up1

prompt msg0
mov si,00h
mov bl,[no+si]
mov ah,00h
up2:
inc si
mov al,[no+si]
add bl,al
jnc down5
inc ah
down5:
dec [b]
jnz up2
mov [d],ah
mov [c],bl
prompt msg3
mov bl,[d]
call display
mov bl,[c]
call display

mov ah,[d]
mov al,[c]
mov bl,05h
div bl
mov bl,al
prompt msg4
call display

prompt msg0
prompt msg2
call take
mov [n1],bl
prompt msg2
call take
mov [n2],bl
mov ah,00h
mov al,[n1]
mov bl,[n2]
mul bl
mov [c],ah
mov [d],al
prompt msg5
mov bl,[c]
call display
mov bl,[d]
call display

mov ah,4ch
int 21h

take proc near
mov ah,01h
int 21h
cmp al,39h
jbe down1
sub al,07h
down1: sub al,30h
mov cl,04h
rol al,cl
mov bl,al

mov ah,01h
int 21h
cmp al,39h
jbe down2
sub al,07h
down2: sub al,30h
add bl,al
ret
take endp

display proc near
mov ch,bl
and bl,0f0h
mov cl,04h
rol bl,cl
cmp bl,09h
jbe down3
add bl,07h
down3: add bl,30h
mov dl,bl
mov ah,02h
int 21h

mov bl,ch
and bl,0fh
cmp bl,09h
jbe down4
add bl,07h
down4: add bl,30h
mov dl,bl
mov ah,02h
int 21h
ret
display endp

code ends
end start
