#include <windows.h>
#include <GL/glut.h>
#include <iostream>
using namespace std;

float xmin=-0.5, ymin=-0.5, xmax=0.5, ymax=0.5;
float x1,y1,x2,y2;
int choice = 1;

int code(float x,float y)
{
    int c=0;
    if(x<xmin) c|=1;
    else if(x>xmax) c|=2;
    if(y<ymin) c|=4;
    else if(y>ymax) c|=8;
    return c;
}

void clip(float x1,float y1,float x2,float y2)
{
    int c1=code(x1,y1), c2=code(x2,y2);

    while(1)
    {
        if((c1==0)&&(c2==0)) break;
        else if(c1 & c2) return;
        else
        {
            float x,y;
            int c = c1 ? c1 : c2;

            if(c & 8){ x=x1+(x2-x1)*(ymax-y1)/(y2-y1); y=ymax; }
            else if(c & 4){ x=x1+(x2-x1)*(ymin-y1)/(y2-y1); y=ymin; }
            else if(c & 2){ y=y1+(y2-y1)*(xmax-x1)/(x2-x1); x=xmax; }
            else { y=y1+(y2-y1)*(xmin-x1)/(x2-x1); x=xmin; }

            if(c==c1){ x1=x; y1=y; c1=code(x1,y1); }
            else { x2=x; y2=y; c2=code(x2,y2); }
        }
    }

    glColor3f(0,1,0);
    glBegin(GL_LINES);
    glVertex2f(x1,y1);
    glVertex2f(x2,y2);
    glEnd();
}

void display()
{
    glClear(GL_COLOR_BUFFER_BIT);

    glColor3f(1,1,1);
    glBegin(GL_LINE_LOOP);
    glVertex2f(xmin,ymin);
    glVertex2f(xmax,ymin);
    glVertex2f(xmax,ymax);
    glVertex2f(xmin,ymax);
    glEnd();


    glColor3f(1,0,0);
    glBegin(GL_LINES);
    glVertex2f(x1,y1);
    glVertex2f(x2,y2);
    glEnd();

    if(choice==1)
        clip(x1,y1,x2,y2);

    glFlush();
}

void init()
{
    glClearColor(0,0,0,1);
    gluOrtho2D(-1,1,-1,1);
}

int main(int argc,char** argv)
{
    cout<<"Enter x1 y1: ";
    cin>>x1>>y1;

    cout<<"Enter x2 y2: ";
    cin>>x2>>y2;

    glutInit(&argc,argv);
    glutCreateWindow("Line Clipping");
    glutInitWindowSize(800,600);

    init();
    glutDisplayFunc(display);


    glutMainLoop();
}
